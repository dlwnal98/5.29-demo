  3.1 API 레이어 분리

  src/core/api/api-client.ts
  import axios, { AxiosRequestConfig, Method } from 'axios';
  import { useAuthStore } from '@/core/auth/auth.store';

  // axios 인스턴스 생성
  export const axiosInstance = axios.create({
    baseURL: import.meta.env.VITE_API_BASE_URL,
    timeout: 30000,
  });

  // 인터셉터는 별도 파일로 분리해도 좋음
  axiosInstance.interceptors.request.use(async (config) => {
    const { accessToken } = useAuthStore.getState();
    if (accessToken) {
      config.headers.Authorization = `Bearer ${accessToken}`;
    }
    return config;
  });

  // 공통 request 함수
  interface RequestOptions<TBody = unknown> {
    pathParams?: Record<string, string | number>;
    queryParams?: Record<string, string>;
    body?: TBody;
  }

  export async function apiRequest<TResponse, TBody = unknown>(
    method: Method,
    url: string,
    options: RequestOptions<TBody> = {}
  ): Promise<TResponse> {
    const { pathParams = {}, queryParams, body } = options;

    // path params 치환
    let finalUrl = url;
    Object.entries(pathParams).forEach(([key, value]) => {
      finalUrl = finalUrl.replace(`{${key}}`, String(value));
    });

    const response = await axiosInstance.request<TResponse>({
      method,
      url: finalUrl,
      params: queryParams,
      data: body,
    });

    return response.data;
  }

  // 편의 함수
  export const api = {
    get: <T>(url: string, options?: RequestOptions) =>
      apiRequest<T>('GET', url, options),
    post: <T, B = unknown>(url: string, options?: RequestOptions<B>) =>
      apiRequest<T, B>('POST', url, options),
    put: <T, B = unknown>(url: string, options?: RequestOptions<B>) =>
      apiRequest<T, B>('PUT', url, options),
    patch: <T, B = unknown>(url: string, options?: RequestOptions<B>) =>
      apiRequest<T, B>('PATCH', url, options),
    delete: <T>(url: string, options?: RequestOptions) =>
      apiRequest<T>('DELETE', url, options),
  };

  import { api } from '@/core/api/api-client';
  import type {
    Resource,
    CreateResourceRequest,
    CreateResourceResponse
  } from '../types/resource.types';

  const ENDPOINTS = {
    OPENAPI_DOC: '/api/v1/plans/{apiId}/openapi',
    RESOURCES: '/api/v1/resources',
    RESOURCE: '/api/v1/resources/{resourceId}',
    RESOURCE_CORS: '/api/v1/resources/{resourceId}/cors',
  } as const;

  export const resourcesApi = {
    getOpenAPIDoc: (apiId: string) =>
      api.get<{ paths: Record<string, unknown> }>(ENDPOINTS.OPENAPI_DOC, {
        pathParams: { apiId },
      }),

    createResource: (data: CreateResourceRequest) =>
      api.post<CreateResourceResponse, CreateResourceRequest>(
        ENDPOINTS.RESOURCES,
        { body: data }
      ),

    deleteResource: (resourceId: string) =>
      api.delete<void>(ENDPOINTS.RESOURCE, {
        pathParams: { resourceId },
      }),

    getCorsSettings: (resourceId: string) =>
      api.get(ENDPOINTS.RESOURCE_CORS, {
        pathParams: { resourceId },
      }),
  };

  src/features/api-management/hooks/use-resources.ts
  import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
  import { resourcesApi } from '../api/resources.api';
  import type { CreateResourceRequest } from '../types/resource.types';

  // Query Keys 중앙 관리
  export const resourceKeys = {
    all: ['resources'] as const,
    openapi: (apiId: string) => [...resourceKeys.all, 'openapi', apiId] as const,
    cors: (resourceId: string) => [...resourceKeys.all, 'cors', resourceId] as const,
  };

  export function useOpenAPIDoc(apiId: string) {
    return useQuery({
      queryKey: resourceKeys.openapi(apiId),
      queryFn: () => resourcesApi.getOpenAPIDoc(apiId),
      enabled: Boolean(apiId),
      staleTime: 5 * 60 * 1000, // 5분
    });
  }

  export function useCreateResource() {
    const queryClient = useQueryClient();

    return useMutation({
      mutationFn: (data: CreateResourceRequest) =>
        resourcesApi.createResource(data),
      onSuccess: () => {
        queryClient.invalidateQueries({ queryKey: resourceKeys.all });
      },
    });
  }

  export function useDeleteResource() {
    const queryClient = useQueryClient();

    return useMutation({
      mutationFn: (resourceId: string) =>
        resourcesApi.deleteResource(resourceId),
      onSuccess: () => {
        queryClient.invalidateQueries({ queryKey: resourceKeys.all });
      },
    });
  }

  3.3 Store 분리

  src/core/auth/auth.store.ts
  import { create } from 'zustand';
  import { persist, createJSONStorage } from 'zustand/middleware';
  import { decodeJWT } from './decode-token';

  interface User {
    userKey: string;
    email: string;
    name: string;
    role: string;
    userId: string;
    organizationId?: string;
    organizationName?: string;
  }

  interface AuthState {
    accessToken: string | null;
    refreshToken: string | null;
    expiresAt: string | null;
    user: User | null;
  }

  interface AuthActions {
    setTokens: (accessToken: string, refreshToken: string, expiresAt: string) => void;
    clearAuth: () => void;
  }

  export const useAuthStore = create<AuthState & AuthActions>()(
    persist(
      (set) => ({
        // State
        accessToken: null,
        refreshToken: null,
        expiresAt: null,
        user: null,

        // Actions
        setTokens: (accessToken, refreshToken, expiresAt) => {
          const user = accessToken ? decodeJWT(accessToken) : null;
          set({ accessToken, refreshToken, expiresAt, user });
        },

        clearAuth: () => {
          set({
            accessToken: null,
            refreshToken: null,
            expiresAt: null,
            user: null
          });
        },
      }),
      {
        name: 'auth-storage',
        storage: createJSONStorage(() => sessionStorage),
      }
    )
  );

  // Selector hooks (컴포넌트 리렌더링 최적화)
  export const useUser = () => useAuthStore((state) => state.user);
  export const useIsAuthenticated = () => useAuthStore((state) => Boolean(state.accessToken));

  import { create } from 'zustand';

  interface DeployState {
    selectedStageId: string | null;
    createdDeploymentId: string | null;
  }

  interface DeployActions {
    setSelectedStage: (id: string | null) => void;
    setCreatedDeploymentId: (id: string | null) => void;
    reset: () => void;
  }

  const initialState: DeployState = {
    selectedStageId: null,
    createdDeploymentId: null,
  };

  export const useDeployStore = create<DeployState & DeployActions>((set) => ({
    ...initialState,
    setSelectedStage: (id) => set({ selectedStageId: id }),
    setCreatedDeploymentId: (id) => set({ createdDeploymentId: id }),
    reset: () => set(initialState),
  }));
